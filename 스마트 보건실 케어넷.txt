        renderAllViews();
      }, (err) => { console.error("Firestore Listen Error:", err); });

      absentColl.onSnapshot((snapshot) => {
        const abss = [];
        snapshot.forEach((doc) => {
          const d = doc.data();
          abss.push({ id: doc.id, ...d, timestamp: d.timestamp?.toDate() || new Date() });
        });
        localData.absents = abss.sort((a, b) => b.timestamp - a.timestamp);
        renderAllViews();
      }, (err) => {});

      alertColl.onSnapshot((snapshot) => {
        const alts = [];
        snapshot.forEach((doc) => {
          const d = doc.data();
          alts.push({ id: doc.id, ...d, timestamp: d.timestamp?.toDate() || new Date() });
        });
        localData.alerts = alts.sort((a, b) => b.timestamp - a.timestamp);
        renderAllViews();
      }, (err) => {});

      profileColl.onSnapshot((snapshot) => {
        const profs = [];
        snapshot.forEach((doc) => {
          const d = doc.data();
          profs.push({ id: doc.id, ...d });
        });
        localData.studentProfiles = profs;
        renderAllViews();
      }, (err) => {});
    }

    function validatePassword() {
      const inputPass = document.getElementById('role-password').value;
      const target = appState.targetPendingRole;

      if (!target) return;

      if (inputPass === ACCESS_PASSWORDS[target]) {
        closePasswordModal();
        switchRole(target);
      } else {
        showModal('shield-alert', '비밀번호가 다릅니다', '지정된 접근 권한 비밀번호를 다시 확인한 뒤 입력해주십시오.', 'error');
        document.getElementById('role-password').value = '';
      }
    }

    // 6. 개인정보 마스킹 처리 유틸리티

    function maskStudentName(name) {
      if (!name) return '';
      const trimName = name.trim();
      if (trimName.length <= 1) return trimName;
      if (trimName.length === 2) {
        return trimName[0] + '*';
      }
      return trimName[0] + '*'.repeat(trimName.length - 2) + trimName[trimName.length - 1];
    }

        timestamp: new Date()

    // 8. 비즈니스 로직 / 상태 변경 함수

    async function updateReceptionStatus(id, newStatus, itemData = null) {
      if (appState.isConnected && user) {
        try {
          const docRef = db.collection('artifacts').doc(appId).collection('public').doc('data').collection('receptions').doc(id);
          if (newStatus === 'delete') {
            await docRef.delete();
          } else {
            await docRef.update({ status: newStatus });
          }
        } catch (e) {}
      } else {
        if (newStatus === 'delete') {
          localData.receptions = localData.receptions.filter(r => r.id !== id);
        } else {
          const item = localData.receptions.find(r => r.id === id);
          if (item) item.status = newStatus;
        }
        renderAllViews();
      }

      if (newStatus === '치료완료' && itemData) {
        sendAutoAlertToTeacher(itemData);
      }
    }

    function renderSymptomCheckboxes() {
      const container = document.getElementById('symptom-container');

        const activeClass = isSelected 
          ? 'ring-2 ring-emerald-500 border-emerald-500 bg-emerald-50 text-emerald-950 font-bold' 
          : 'bg-white border-slate-200 text-slate-700 hover:bg-slate-50 hover:border-slate-300';

        const btn = document.createElement('button');

      } else {

      renderSymptomCheckboxes();
    }

    // ----------------------------------------------------
    function renderAllViews() {
      if (appState.role === 'landing') return;

      const waitingCount = localData.receptions.filter(r => r.status === '대기').length;
      const completeCount = localData.receptions.filter(r => r.status === '치료완료').length;
      const absentCount = localData.absents.length;
      const alertCount = localData.alerts.length;

      const statWaiting = document.getElementById('stat-waiting-count');
      const statComplete = document.getElementById('stat-complete-count');
      const statAbsent = document.getElementById('stat-absent-count');
      const statAlert = document.getElementById('stat-alert-count');

      if (statWaiting) statWaiting.innerText = `${waitingCount}명`;
      if (statComplete) statComplete.innerText = `${completeCount}명`;
      if (statAbsent) statAbsent.innerText = `${absentCount}건`;
      if (statAlert) statAlert.innerText = `${alertCount}건`;

      const kioskWaitContainer = document.getElementById('kiosk-waiting-list');
      if (kioskWaitContainer) {
        const curWaitingList = localData.receptions.filter(r => r.status === '대기');
        if (curWaitingList.length === 0) {
          kioskWaitContainer.innerHTML = `<p class="text-xs text-slate-400 py-6 text-center">대기 중인 학생이 없습니다.</p>`;
        } else {
          kioskWaitContainer.innerHTML = curWaitingList.map((item, index) => {
            const queueNum = curWaitingList.length - index;
            const maskedName = maskStudentName(item.name);
            return `
              <div class="flex items-center justify-between p-3 bg-slate-50 border border-slate-100 rounded-2xl shadow-sm">
                <div class="flex items-center gap-2.5">
                  <span class="text-xs font-black text-emerald-600 bg-emerald-50 px-2.5 py-1 rounded-xl">대기 ${queueNum}번</span>
                  <div>
                    <h5 class="text-sm font-bold text-slate-800">${item.grade}학년 ${item.class}반 ${maskedName}</h5>
                    <p class="text-[10px] text-slate-400 mt-0.5">${item.symptoms.slice(0, 2).join(', ')}</p>
                  </div>
                </div>
                <span class="text-[10px] bg-emerald-100 text-emerald-800 px-2 py-1 rounded-full font-bold">진료 준비중</span>
              </div>
            `;
          }).join('');
        }
      }

      const nurseListContainer = document.getElementById('nurse-reception-list');
      if (nurseListContainer) {
        if (localData.receptions.length === 0) {
          nurseListContainer.innerHTML = `<p class="text-center text-slate-400 py-12">현재 접수되거나 대기중인 학생이 없습니다.</p>`;
        } else {
          nurseListContainer.innerHTML = localData.receptions.map(item => {
            const isWaiting = item.status === '대기';
            const isCaring = item.status === '치료중';
            
            const matchedProfile = localData.studentProfiles.find(p => 
              String(p.grade) === String(item.grade) && 
              String(p.class) === String(item.class) && 
              String(p.name) === String(item.name)
            );
            
            const isFavorite = matchedProfile?.isFavorite || false;
            const healthAlert = matchedProfile?.notes || '';

            let cardBorder = 'border-slate-100';
            let statusBadge = '';
            if (isWaiting) {
              cardBorder = 'border-l-4 border-l-emerald-500';
              statusBadge = `<span class="bg-emerald-100 text-emerald-800 text-xs px-2.5 py-1 rounded-full font-bold">진료 대기</span>`;
            } else if (isCaring) {
              cardBorder = 'border-l-4 border-l-amber-500';
              statusBadge = `<span class="bg-amber-100 text-amber-800 text-xs px-2.5 py-1 rounded-full font-bold">처치/휴식 중</span>`;
            } else {
              cardBorder = 'border-l-4 border-l-slate-300 opacity-60';
              statusBadge = `<span class="bg-slate-100 text-slate-600 text-xs px-2.5 py-1 rounded-full font-semibold">처치 완료</span>`;
            }

            return `
              <div class="bg-white border ${cardBorder} p-5 rounded-2xl shadow-sm space-y-4 hover:shadow-md transition-all">
                <div class="flex justify-between items-start">
                  <div>
                    <div class="flex items-center gap-2">
                      ${isFavorite ? `<span class="text-amber-500 text-xl font-bold" title="자주 방문 학생(즐겨찾기)">★</span>` : ''}
                      <span class="text-lg font-black text-slate-900">${item.grade}학년 ${item.class}반 ${item.number}번 <span class="text-emerald-600">${item.name}</span></span>
                      ${statusBadge}
                    </div>
                    <p class="text-xs text-slate-400 mt-1">${formatTime(item.timestamp)}</p>
                  </div>
                  
                  <div class="flex items-center gap-1.5">
                    ${isWaiting ? `
                      <button onclick="updateReceptionStatus('${item.id}', '치료중')" class="text-xs bg-amber-500 hover:bg-amber-600 text-white font-bold px-3 py-2 rounded-lg flex items-center gap-1">
                        <i data-lucide="play" class="w-3 h-3"></i> 치료 시작
                      </button>
                    ` : ''}
                    ${isWaiting || isCaring ? `
                      <button onclick='updateReceptionStatus(\"${item.id}\", \"치료완료\", ${JSON.stringify(item).replace(/"/g, '&quot;')})' class="text-xs bg-emerald-500 hover:bg-emerald-600 text-white font-bold px-3 py-2 rounded-lg flex items-center gap-1">
                        <i data-lucide="check" class="w-3 h-3"></i> 조치 완료 (담임 연동)
                      </button>
                    ` : ''}
                    <button onclick="updateReceptionStatus('${item.id}', 'delete')" class="text-xs bg-rose-50 hover:bg-rose-100 text-rose-600 font-bold p-2 rounded-lg" title="기록 지우기">
                      <i data-lucide="trash-2" class="w-4 h-4"></i>
                    </button>
                  </div>
                </div>

                <div class="flex flex-wrap gap-1.5">
                  ${item.symptoms.map(s => `<span class="bg-slate-100 text-slate-700 text-xs px-2.5 py-1 rounded-lg border border-slate-200">${s}</span>`).join('')}
                </div>

                ${healthAlert ? `
                  <div class="bg-rose-50 border border-rose-100 p-3 rounded-xl text-xs text-rose-800 flex items-start gap-2">
                    <i data-lucide="alert-triangle" class="w-4 h-4 text-rose-600 shrink-0 mt-0.5"></i>
                    <div>
                      <strong class="font-bold">보건대장 등록 건강 유의사항:</strong> ${healthAlert}
                    </div>
                  </div>
                ` : ''}

                ${item.detail ? `
                  <div class="bg-slate-50 p-3 rounded-xl border border-slate-100/80 text-sm text-slate-600">
                    <span class="font-bold text-slate-800 text-xs block mb-1">학생 진술 및 작성 내용:</span>
                    ${item.detail}
                  </div>
                ` : ''}
              </div>
            `;
          }).join('');
        }
      }

      const nurseProfileTable = document.getElementById('nurse-profile-table-body');
      if (nurseProfileTable) {
        if (localData.studentProfiles.length === 0) {
          nurseProfileTable.innerHTML = `
            <tr>
              <td colspan="5" class="px-4 py-4 text-center text-slate-400">등록된 유의학생 정보가 없습니다.</td>
            </tr>
          `;
        } else {
          nurseProfileTable.innerHTML = localData.studentProfiles.map(p => `
            <tr class="hover:bg-slate-50/50">
              <td class="px-4 py-2 text-slate-600 font-medium">${p.grade}-${p.class}-${p.number}</td>
              <td class="px-4 py-2 font-bold text-slate-900">${p.name}</td>
              <td class="px-4 py-2 text-rose-700 font-semibold">${p.notes}</td>
              <td class="px-4 py-2 text-center text-amber-500 font-black">${p.isFavorite ? '★ 자주 방문' : '일반'}</td>
              <td class="px-4 py-2 text-center">
                <button onclick="deleteStudentProfile('${p.id}')" class="text-slate-400 hover:text-red-500">
                  <i data-lucide="trash" class="w-3.5 h-3.5 mx-auto"></i>
                </button>
              </td>
            </tr>
          `).join('');
        }
      }

      const nurseAbsentContainer = document.getElementById('nurse-absent-list');
      if (nurseAbsentContainer) {
        if (localData.absents.length === 0) {
          nurseAbsentContainer.innerHTML = `<p class="text-center text-slate-400 py-6 text-sm">확인되지 않은 부재중 쪽지가 없습니다.</p>`;
        } else {
          nurseAbsentContainer.innerHTML = localData.absents.map(item => `
            <div class="bg-indigo-50/50 border border-indigo-100 rounded-2xl p-4 space-y-2 relative">
              <div class="flex justify-between items-start">
                <span class="font-bold text-indigo-950 text-sm">${item.student}</span>

                  <i data-lucide="x-circle" class="w-4 h-4"></i>
                </button>

              <p class="text-xs text-indigo-900 font-medium">${item.message}</p>

            </div>
          `).join('');
        }
      }

      const teacherFeedContainer = document.getElementById('teacher-feed-container');
      if (teacherFeedContainer) {
        const myClassFeed = localData.alerts.filter(item => String(item.grade) === String(appState.teacherGrade) && String(item.class) === String(appState.teacherClass));

        const myVisitToday = localData.receptions.filter(r => String(r.grade) === String(appState.teacherGrade) && String(r.class) === String(appState.teacherClass)).length;
        const teacherClassVisitCount = document.getElementById('teacher-class-visit-count');
        if (teacherClassVisitCount) {

          teacherFeedContainer.innerHTML = `
            <div class="text-center py-16 text-slate-400">

              <p class="font-semibold text-sm">해당 학급과 동기화된 새로운 보건실 알림 정보가 없습니다.</p>
            </div>

        } else {
          teacherFeedContainer.innerHTML = myClassFeed.map(item => {
            const isDocRequest = item.action.includes('[담임 요청]');

            const matchedProfile = localData.studentProfiles.find(p => 
              String(p.grade) === String(appState.teacherGrade) && 
              String(p.class) === String(appState.teacherClass) && 
              String(p.name) === String(item.student)
            );
            
            const hasNotes = matchedProfile?.notes || '';

            let themeBg = 'bg-slate-50 border-slate-100';
            let themeBadge = '<span class="bg-slate-100 text-slate-700 text-xs px-2 py-0.5 rounded-full font-bold">전반 알람</span>';
            
            if (isDocRequest) {
              themeBg = 'bg-blue-50/50 border-blue-100';
              themeBadge = '<span class="bg-blue-100 text-blue-800 text-xs px-2.5 py-1 rounded-full font-bold">전달사항 요청됨</span>';
            } else if (isCared) {
              themeBg = 'bg-emerald-50/50 border-emerald-100';
              themeBadge = '<span class="bg-emerald-100 text-emerald-800 text-xs px-2.5 py-1 rounded-full font-bold">처치 최종 복귀</span>';
            }

            return `
              <div class="border p-4 rounded-2xl ${themeBg} space-y-2">
                <div class="flex justify-between items-center">
                  <div class="flex items-center gap-2">
                    <span class="font-extrabold text-slate-900">${item.student} 학생</span>

                  </div>
                  <span class="text-[10px] text-slate-400">${formatTime(item.timestamp)}</span>
                </div>
                <p class="text-sm text-slate-700 font-medium leading-relaxed">${item.action}</p>
                
                ${hasNotes ? `
                  <div class="bg-rose-50 border border-rose-100 p-2.5 rounded-xl text-xs text-rose-800 mt-2 flex items-start gap-1.5">
                    <i data-lucide="alert-triangle" class="w-3.5 h-3.5 text-rose-600 shrink-0 mt-0.5"></i>
                    <div>
                      <strong class="font-bold">이 학생의 건강 특이사항:</strong> ${hasNotes} (체크 요망)
                    </div>
                  </div>
                ` : ''}
              </div>
            `;
          }).join('');
        }
      }

      lucide.createIcons();
    }

    function formatTime(date) {
      if (!date) return '';
      const hrs = String(date.getHours()).padStart(2, '0');
      const mins = String(date.getMinutes()).padStart(2, '0');
      return `${hrs}:${mins}`;
    }

    function updateStatusBanner(text, type) {
      const banner = document.getElementById('status-banner');
      const textElem = document.getElementById('status-text');
      const dot = document.getElementById('status-dot');

      banner.className = "border-b px-6 py-2 flex items-center justify-between text-xs transition-all ";
      if (type === 'success') {
        banner.classList.add('bg-emerald-50', 'border-emerald-200', 'text-emerald-800');
        dot.className = "w-2 h-2 rounded-full bg-emerald-500 animate-pulse";
        textElem.innerText = text;
      } else if (type === 'warning') {
        banner.classList.add('bg-amber-50', 'border-amber-200', 'text-amber-800');
        dot.className = "w-2 h-2 rounded-full bg-amber-500 animate-pulse";
        textElem.innerText = text;
      }
    }

    function retryConnection() {
      updateStatusBanner('서버 재연결 시도 중...', 'warning');
      initFirebase();
    }

    function showModal(icon, title, desc, status) {
      const modal = document.getElementById('alert-modal');
      const iconContainer = document.getElementById('modal-icon-container');
      const titleElem = document.getElementById('modal-title');
      const descElem = document.getElementById('modal-desc');

      titleElem.innerText = title;
      descElem.innerText = desc;

      iconContainer.className = "w-16 h-16 rounded-full mx-auto flex items-center justify-center ";
      if (status === 'success') {
        iconContainer.classList.add('bg-emerald-50', 'text-emerald-500');
        iconContainer.innerHTML = `<i data-lucide="check-circle" class="w-8 h-8"></i>`;
      } else {
        iconContainer.classList.add('bg-red-50', 'text-red-500');
        iconContainer.innerHTML = `<i data-lucide="alert-circle" class="w-8 h-8"></i>`;
      }

      lucide.createIcons();
      modal.classList.remove('opacity-0', 'pointer-events-none');
      modal.querySelector('div').classList.remove('scale-95');
    }

    function closeModal() {
      const modal = document.getElementById('alert-modal');
      modal.classList.add('opacity-0', 'pointer-events-none');
      modal.querySelector('div').classList.add('scale-95');
    }
